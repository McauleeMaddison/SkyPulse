    static Mesh Support(int world,int variant)
    {
        var prefab=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Editor/SkyPulseBackgroundWorkshop/Prefabs/world-"+world.ToString("00")+"-civic.prefab");
        var parts=prefab.GetComponentsInChildren<MeshFilter>();
        var buildings=parts.Where(p=>p.name.Contains("Building")||p.name.Contains("Castle tower")||p.name.Contains("Pavilion")||p.name.Contains("Gatehouse")).OrderBy(p=>p.sharedMesh.bounds.size.y).ToArray();
        var chosen=buildings.Skip(variant).Take(variant==0?1:2).ToArray();if(chosen.Length==0)chosen=buildings.Take(1).ToArray();
        var structural=parts.Where(p=>p.name.Contains("Cliff island")||p.name.Contains("foundation")||p.name.Contains("Promenade")||p.name.Contains("Garden terrace"));
        var selection=structural.Concat(chosen).Distinct().ToArray();if(selection.Length<2)throw new Exception("Incomplete source support district: "+world);
        var mesh=new Mesh{name="Study world "+world+" support quarter "+variant,indexFormat=IndexFormat.UInt32};
        mesh.CombineMeshes(selection.Select(p=>new CombineInstance{mesh=p.sharedMesh,transform=prefab.transform.worldToLocalMatrix*p.transform.localToWorldMatrix}).ToArray(),true,true);mesh.RecalculateBounds();owned.Add(mesh);return mesh;
    }

    static void RelayStyle(SkyPulseNativeGame g,int option)
    {
        var e=(SkyPulseRelayEnvironment)Get(g,"relayEnvironment");var scene=(Transform)Get(e,"scene");
        foreach(var tr in scene.GetComponentsInChildren<Transform>(true))
        {
            string n=tr.name;
            if(n.Contains("recycler")||n.Contains("launch hub")||n.Contains("Overhead optical")||n.Contains("Inner orbital")||n.Contains("Outer orbital")||n.Contains("Lower distant moon")||n.Contains("upper concourse")||n.Contains("Upper transit"))tr.gameObject.SetActive(false);
            if(n.Contains("Distant service spire")&&(n.EndsWith("2")||n.EndsWith("4")))tr.gameObject.SetActive(false);
            if(n.Contains("maintenance drone")&&n.EndsWith("1"))tr.gameObject.SetActive(false);
            if(option==2&&(n.Contains("transit bridge")||n.Contains("Transit bridge")||n.StartsWith("West transfer")||n.StartsWith("West ")&&n.Contains("concourse")))tr.gameObject.SetActive(false);
        }
        var planet=scene.GetComponentsInChildren<Transform>(true).FirstOrDefault(t=>t.name=="Distant blue planet");
        if(planet!=null){planet.localPosition=new Vector3(8,14,96);planet.localScale=Vector3.one*(option==1?10.5f:12f);}
        // Recolour only station hardware, never locked/unlocked gameplay colours
        // or planetary shaders. Preserve all normal maps and real surface detail.
        Color alloy=option==1?new Color(.70f,.76f,.79f):option==2?new Color(.17f,.25f,.33f):new Color(.32f,.43f,.53f);
        foreach(var filter in scene.GetComponentsInChildren<MeshFilter>(true))
        {
            var renderer=filter.GetComponent<MeshRenderer>();if(renderer==null||filter.sharedMesh==null)continue;string shader=renderer.sharedMaterial.shader.name;
            if(!(shader.Contains("Orbital Station Hardware")||shader.Contains("Flight Architecture")))continue;
            renderer.sharedMaterial.SetColor("_Tint",option==1?new Color(.45f,.80f,1):new Color(.24f,.69f,.96f));
            if(!filter.sharedMesh.isReadable)continue;
            var mesh=UnityEngine.Object.Instantiate(filter.sharedMesh);owned.Add(mesh);filter.sharedMesh=mesh;var colours=mesh.colors;
            for(int i=0;i<colours.Length;i++)if(colours[i].a<.25f){float v=Mathf.Clamp(.68f+colours[i].grayscale*.60f,.70f,1.1f);float alpha=colours[i].a;colours[i]=alloy*v;colours[i].a=alpha;}mesh.colors=colours;
        }
        // The large decorative glow competes with the crisp hoop; soften it.
        var mission=(Transform)Get(g,"missionWorld");foreach(var sr in mission.GetComponentsInChildren<SpriteRenderer>())if(sr.name=="Station atmosphere"||sr.name=="Gold horizon"){var c=sr.color;c.a*=.30f;sr.color=c;}
        e.WorldCamera.Render();
    }
    static void BoardStyle(SkyPulseNativeGame g,int option)
    {
        var board=(SkyPulseHeistBoard)Get(g,"blackoutBoard");var scene=(Transform)Get(board,"scene");
        ((Transform)Get(board,"projectionField")).gameObject.SetActive(false);
        ((MeshRenderer)Get(board,"channelRenderer")).enabled=option==2;
        ((MeshRenderer)Get(board,"platenRenderer")).enabled=option!=0;
        foreach(var material in (List<Material>)Get(board,"materials"))
        {
            if(!material.HasProperty("_BaseColor"))continue;
            if(material.name.Contains("mirror prism"))material.SetColor("_BaseColor",new Color(.33f,.58f,.74f,.88f));
            if(material.name.Contains("optical glass"))material.SetColor("_BaseColor",new Color(.16f,.45f,.62f,.80f));
            if(material.name.Contains("worktop"))material.SetColor("_BaseColor",option==1?new Color(.47f,.54f,.58f):option==2?new Color(.025f,.038f,.058f):new Color(.040f,.065f,.10f));

            if(material.name.Contains("titanium edges"))material.SetColor("_BaseColor",option==1?new Color(.62f,.69f,.72f):new Color(.30f,.41f,.49f));

        foreach(var renderer in scene.GetComponentsInChildren<MeshRenderer>(true))
        {
            if(renderer.name=="Pooled charge impulse"||renderer.name=="Soft floating contact")renderer.enabled=false;
            if(renderer.name=="Recessed luminous circuit gasket")renderer.transform.localScale=Vector3.one*(option==1?.83f:.62f);

        // Lower background ornament and retain the actual restoration objective.
        var root=(GameObject)Get(g,"blackoutScreen");foreach(var image in root.GetComponentsInChildren<Image>(true))if(image.name.Contains("backdrop")||image.name.Contains("Backdrop"))image.color=option==1?new Color(.025f,.050f,.075f,1):new Color(.006f,.014f,.030f,1);

        }

    }
